! ===== SW1 (HQ access switch, 2960) =====
enable
configure terminal
hostname SW1
no ip domain-lookup
ip domain-name lab.local
service password-encryption
enable secret ChangeMe!23
username admin privilege 15 secret ChangeMe!23
banner motd # Authorized access only #

vlan 10
 name STAFF
vlan 20
 name IT
vlan 30
 name GUEST
vlan 99
 name MGMT
vlan 998
 name PARKING
vlan 999
 name NATIVE

! --- Access ports with port security ---
interface range fa0/1 - 5
 switchport mode access
 switchport access vlan 10
 spanning-tree portfast
 spanning-tree bpduguard enable
 switchport port-security
 switchport port-security maximum 2
 switchport port-security violation restrict
 switchport port-security mac-address sticky
interface range fa0/6 - 10
 switchport mode access
 switchport access vlan 20
 spanning-tree portfast
 spanning-tree bpduguard enable
 switchport port-security
 switchport port-security maximum 2
 switchport port-security violation restrict
 switchport port-security mac-address sticky
interface range fa0/11 - 15
 switchport mode access
 switchport access vlan 30
 spanning-tree portfast
 spanning-tree bpduguard enable
 switchport port-security
 switchport port-security maximum 2
 switchport port-security violation restrict
 switchport port-security mac-address sticky

! --- Unused ports: parked and shut ---
interface range fa0/16 - 24, gi0/2
 switchport mode access
 switchport access vlan 998
 shutdown

! --- Trunk to R1 ---
interface gi0/1
 description Trunk to R1
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,99,999
 switchport nonegotiate

! --- Management ---
interface vlan 99
 ip address 192.168.99.2 255.255.255.0
 no shutdown
ip default-gateway 192.168.99.1
crypto key generate rsa general-keys modulus 2048
ip ssh version 2
ip access-list standard MGMT_ONLY
 permit 192.168.20.0 0.0.0.255
line vty 0 15
 access-class MGMT_ONLY in
 transport input ssh
 login local

 logging synchronous
 exec-timeout 10 0
end
