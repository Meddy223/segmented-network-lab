! ===== R1 (HQ router, ISR 4331) =====
enable
configure terminal
hostname R1
no ip domain-lookup
ip domain-name lab.local
service password-encryption
enable secret ChangeMe!23
username admin privilege 15 secret ChangeMe!23
banner motd # Authorized access only #

! --- WAN link to R2 ---
interface g0/0/0
 description Link to R2 (Branch)
 ip address 10.0.0.1 255.255.255.252
 no shutdown

! --- Router-on-a-stick trunk to SW1 ---
interface g0/0/1
 description Trunk to SW1
 no shutdown
interface g0/0/1.10
 description STAFF
 encapsulation dot1Q 10
 ip address 192.168.10.1 255.255.255.0
interface g0/0/1.20
 description IT
 encapsulation dot1Q 20
 ip address 192.168.20.1 255.255.255.0
interface g0/0/1.30
 description GUEST
 encapsulation dot1Q 30
 ip address 192.168.30.1 255.255.255.0
interface g0/0/1.99
 description MGMT
 encapsulation dot1Q 99
 ip address 192.168.99.1 255.255.255.0
interface g0/0/1.999
 description NATIVE (unused)
 encapsulation dot1Q 999 native

! --- DHCP (first 10 addresses reserved for static devices) ---
ip dhcp excluded-address 192.168.10.1 192.168.10.10
ip dhcp excluded-address 192.168.20.1 192.168.20.10
ip dhcp excluded-address 192.168.30.1 192.168.30.10
ip dhcp pool STAFF
 network 192.168.10.0 255.255.255.0
 default-router 192.168.10.1
 dns-server 192.168.20.10
ip dhcp pool IT
 network 192.168.20.0 255.255.255.0
 default-router 192.168.20.1
 dns-server 192.168.20.10
! No dns-server here: GUEST_IN blocks the internal DNS server, so guests test by IP
ip dhcp pool GUEST
 network 192.168.30.0 255.255.255.0
 default-router 192.168.30.1

! --- OSPF ---
router ospf 1
 router-id 1.1.1.1
 passive-interface default
 no passive-interface g0/0/0
 network 10.0.0.0 0.0.0.3 area 0
 network 192.168.10.0 0.0.0.255 area 0
 network 192.168.20.0 0.0.0.255 area 0
 network 192.168.30.0 0.0.0.255 area 0
 network 192.168.99.0 0.0.0.255 area 0

! --- ACLs ---
! Staff: DNS + web to the IT server only; no other IT or MGMT access
! (echo-reply/established let replies to IT-initiated traffic back through)
ip access-list extended STAFF_IN
 permit udp any eq bootpc any eq bootps
 permit udp 192.168.10.0 0.0.0.255 host 192.168.20.10 eq domain
 permit tcp 192.168.10.0 0.0.0.255 host 192.168.20.10 eq www
 permit icmp 192.168.10.0 0.0.0.255 192.168.20.0 0.0.0.255 echo-reply
 permit tcp 192.168.10.0 0.0.0.255 192.168.20.0 0.0.0.255 established
 deny ip 192.168.10.0 0.0.0.255 192.168.20.0 0.0.0.255
 deny ip 192.168.10.0 0.0.0.255 192.168.99.0 0.0.0.255
 permit ip 192.168.10.0 0.0.0.255 any
! Guest: internet only, nothing internal (replies to IT-initiated traffic allowed)
ip access-list extended GUEST_IN
 permit udp any eq bootpc any eq bootps
 permit icmp 192.168.30.0 0.0.0.255 192.168.20.0 0.0.0.255 echo-reply
 permit tcp 192.168.30.0 0.0.0.255 192.168.20.0 0.0.0.255 established
 deny ip 192.168.30.0 0.0.0.255 192.168.0.0 0.0.255.255
 deny ip 192.168.30.0 0.0.0.255 10.0.0.0 0.255.255.255
 permit ip 192.168.30.0 0.0.0.255 any
interface g0/0/1.10
 ip access-group STAFF_IN in
interface g0/0/1.30
 ip access-group GUEST_IN in

! --- SSH management, IT VLAN only ---
crypto key generate rsa general-keys modulus 2048
ip ssh version 2
ip access-list standard MGMT_ONLY
 permit 192.168.20.0 0.0.0.255
line vty 0 4
 access-class MGMT_ONLY in
 transport input ssh
 login local
line con 0
 login local
 logging synchronous
 exec-timeout 10 0
end
copy running-config startup-config
