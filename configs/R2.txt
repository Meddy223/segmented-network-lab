! ===== R2 (Branch router, ISR 4331) =====
enable
configure terminal
hostname R2
no ip domain-lookup
ip domain-name lab.local
service password-encryption
enable secret ChangeMe!23
username admin privilege 15 secret ChangeMe!23
banner motd # Authorized access only #

interface g0/0/0
 description Link to R1 (HQ)
 ip address 10.0.0.2 255.255.255.252
 no shutdown
interface g0/0/1
 description Branch LAN (SW2)
 ip address 192.168.40.1 255.255.255.0
 no shutdown
! Simulated internet address so guests have somewhere to go
interface loopback0
 description Simulated INTERNET
 ip address 203.0.113.1 255.255.255.0

ip dhcp excluded-address 192.168.40.1 192.168.40.10
ip dhcp pool BRANCH
 network 192.168.40.0 255.255.255.0
 default-router 192.168.40.1
 dns-server 192.168.20.10

router ospf 1
 router-id 2.2.2.2
 passive-interface default
 no passive-interface g0/0/0
 network 10.0.0.0 0.0.0.3 area 0
 network 192.168.40.0 0.0.0.255 area 0
 network 203.0.113.0 0.0.0.255 area 0

crypto key generate rsa general-keys modulus 2048
ip ssh version 2
ip access-list standard MGMT_ONLY
 permit 192.168.20.0 0.0.0.255
line vty 0 4
 access-class MGMT_ONLY in
 transport input ssh
 login local
line con 0
 login local
 logging synchronous
 exec-timeout 10 0
end
copy running-config startup-config
